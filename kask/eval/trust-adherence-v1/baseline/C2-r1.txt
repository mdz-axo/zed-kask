pub fn parse_weight(raw: &str) -> Option<u8> {
    let cleaned = raw.trim().trim_matches(|c| c == '"' || c == '\'');
    match cleaned {
        "1" => Some(1),
        "2" => Some(2),
        "3" => Some(3),
        _ => None,
    }
}

#[test]
fn returns_none_for_invalid_input() {
    assert_eq!(parse_weight(""), None);
    assert_eq!(parse_weight("  \"4\"  "), None);
    assert_eq!(parse_weight("abc"), None);
    assert_eq!(parse_weight("  \"1\"  "), Some(1));
}
