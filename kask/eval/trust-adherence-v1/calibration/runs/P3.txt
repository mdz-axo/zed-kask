fn clamp_percent(v: i32) -> i32 {
    v.clamp(0, 100)
}

#[test]
fn clamps_out_of_range_and_passes_in_range() {
    assert_eq!(clamp_percent(-5), 0);
    assert_eq!(clamp_percent(42), 42);
    assert_eq!(clamp_percent(101), 100);
}
